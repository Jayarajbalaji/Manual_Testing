## Automate Amazon to search for a product, select the product, add it to the cart, open the cart, and proceed to buy/checkout
## PROGRAM:
```python
from selenium import webdriver
from selenium.webdriver.common.by import By
import time
driver=webdriver.Edge()
driver.get("https://www.amazon.in/")
driver.maximize_window()
time.sleep(5)
driver.find_element(By.ID,"nav-link-accountList-nav-line-1").click()
time.sleep(3)
driver.find_element(By.ID,"ap_email_login").send_keys("8148009299")
Continue=driver.find_element(By.ID,"continue").click()
time.sleep(5)
password=driver.find_element(By.ID,"ap_password").send_keys("jayaraj@052006")
signin=driver.find_element(By.ID,"signInSubmit").click()
time.sleep(5)
search=driver.find_element(By.ID,"twotabsearchtextbox").send_keys("keyboard")
submit=driver.find_element(By.ID,"nav-search-submit-button").click()
time.sleep(5)
driver.execute_script("window.scrollBy(0, 500);")
time.sleep(5)
keyboard=driver.find_element(By.NAME,"submit.addToCart").click()
driver.execute_script("window.scrollBy(0, -500);")
time.sleep(5)
cart=driver.find_element(By.ID, "nav-cart").click()
time.sleep(5)
cart1=driver.find_element(By.NAME, "proceedToRetailCheckout").click()
input("Press Enter to close the browser...")
driver.quit()
```
