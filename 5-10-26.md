## SauceDemo Login Automation
### Automate the SauceDemo login process by entering valid username and password credentials using Selenium WebDriver.
```python
from selenium import webdriver
from selenium.webdriver.common.by import By

driver=webdriver.Chrome()
driver.get("https://www.saucedemo.com/")

username=driver.find_element(By.ID,"user-name")
password=driver.find_element(By.NAME,"password")
login=driver.find_element(By.ID,"login-button")

username.send_keys("standard_user")
password.send_keys("secret_sauce")

print(
    username.get_attribute("placeholder")
)
print(
    login.is_enabled()
)
print(
    username.is_displayed()
)
login.click()
driver.quit()

```

## Flipkart Login Automation
### Automate Flipkart login by entering a mobile number, submitting OTP, and completing the login process using Selenium WebDriver.
```python
from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC

driver = webdriver.Edge()

driver.get("https://www.flipkart.com/")

wait = WebDriverWait(driver, 10)

login = wait.until(
    EC.element_to_be_clickable((By.CLASS_NAME, "wszdrO"))
)
login.click()

phone = wait.until(
    EC.visibility_of_element_located((By.CLASS_NAME, "jwCbxy"))
)
phone.send_keys("1234567890")

button = wait.until(
    EC.element_to_be_clickable(
        (By.CSS_SELECTOR, ".xqOMQN.FFO0ui")
    )
)
button.click()

otp = wait.until(
    EC.visibility_of_element_located((By.CLASS_NAME, "S1KmoO"))
)

otp.send_keys(int(input("Enter OTP:")))

button1 = wait.until(
    EC.element_to_be_clickable(
        (By.CSS_SELECTOR, ".xqOMQN.FFO0ui")
    )
)
button1.click()

input("Press Enter to close the browser...")

driver.quit()
```
